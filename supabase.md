## Modelo recomendado

Para produção, o login deve usar o **Supabase Auth**. A senha não deve ser gravada na tabela `public.users`; o Supabase mantém o hash de senha em `auth.users`.

Para preservar a compatibilidade com o código atual, a tabela de perfil se chama `users` e usa o mesmo `id` do usuário autenticado no Supabase.

## SQL inicial

O conteúdo abaixo documenta o mesmo modelo do arquivo [jornal.sql](/home/drezin/Documents/Elza/jornal.sql). Execute o arquivo `jornal.sql` no SQL Editor do Supabase. Recomenda-se executar em um projeto vazio ou revisar cada bloco antes de aplicar em um banco existente.

```sql
create extension if not exists pgcrypto;

create type public.user_profile as enum ('estudante', 'professor', 'coordenador', 'diretor', 'admin');
create type public.suggestion_status as enum ('recebido', 'em_analise', 'respondido', 'arquivado');
create type public.article_status as enum ('rascunho', 'aguardando_coordenacao', 'em_revisao', 'aprovado', 'publicado', 'recusado');

create table public.users (
  id uuid primary key references auth.users(id) on delete cascade,
  nome varchar(150) not null,
  email varchar(255) not null unique,
  perfil public.user_profile not null default 'estudante',
  matricula varchar(50),
  serie varchar(20),
  turma varchar(20),
  telefone varchar(30),
  ativo boolean not null default true,
  ultimo_login_at timestamptz,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now(),
  deleted_at timestamptz
);

create table public.media_files (
  id bigint generated by default as identity primary key,
  owner_id uuid references public.users(id) on delete set null,
  file_name varchar(255) not null,
  original_name varchar(255) not null,
  storage_path text not null unique,
  public_url text,
  mime_type varchar(100) not null,
  size_bytes bigint not null check (size_bytes >= 0),
  alt_text varchar(255),
  created_at timestamptz not null default now(),
  deleted_at timestamptz
);

create table public.articles (
  id bigint generated by default as identity primary key,
  author_id uuid references public.users(id) on delete set null,
  autor_nome varchar(150) not null,
  categoria varchar(100) not null check (categoria in ('noticia_geral', 'artigo_cientifico', 'ser_onu', 'cultura_afro')),
  titulo varchar(255) not null,
  slug varchar(255) unique,
  resumo text not null,
  conteudo text not null,
  imagem_url text,
  status public.article_status not null default 'aguardando_coordenacao',
  motivo_recusa text,
  featured boolean not null default false,
  published_at timestamptz,
  published_by uuid references public.users(id) on delete set null,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now(),
  deleted_at timestamptz
);

create table public.events (
  id bigint generated by default as identity primary key,
  categoria varchar(100),
  titulo varchar(255) not null,
  slug varchar(255) unique,
  descricao text,
  data_inicio timestamptz not null,
  data_fim timestamptz,
  local varchar(255) not null,
  imagem_id bigint references public.media_files(id) on delete set null,
  inscricao_aberta boolean not null default false,
  inscricao_url text,
  status varchar(50) not null default 'publicado' check (status in ('rascunho', 'publicado', 'arquivado')),
  created_by uuid references public.users(id) on delete set null,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now(),
  deleted_at timestamptz,
  check (data_fim is null or data_fim >= data_inicio)
);

create table public.suggestions (
  id bigint generated by default as identity primary key,
  anonimo boolean not null default false,
  nome varchar(150),
  email varchar(255),
  perfil varchar(50),
  categoria varchar(100) not null default 'outro',
  titulo varchar(255) not null,
  mensagem text not null,
  status public.suggestion_status not null default 'recebido',
  responsavel_id uuid references public.users(id) on delete set null,
  resposta text,
  respondido_at timestamptz,
  ip_hash varchar(255),
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now(),
  deleted_at timestamptz
);

create table public.audit_logs (
  id bigint generated by default as identity primary key,
  user_id uuid references public.users(id) on delete set null,
  acao varchar(100) not null,
  entidade varchar(100) not null,
  entidade_id bigint,
  dados_novos jsonb,
  ip_hash varchar(255),
  created_at timestamptz not null default now()
);
```

## Índices e atualização automática

```sql
create index if not exists articles_status_idx on public.articles(status) where deleted_at is null;
create index if not exists articles_author_idx on public.articles(author_id);
create index if not exists events_start_idx on public.events(data_inicio) where deleted_at is null;
create index if not exists suggestions_status_idx on public.suggestions(status) where deleted_at is null;
create index if not exists audit_logs_entity_idx on public.audit_logs(entidade, entidade_id);

create or replace function public.set_updated_at()
returns trigger
language plpgsql
as $$
begin
  new.updated_at = now();
  return new;
end;
$$;

create trigger users_set_updated_at
before update on public.users
for each row execute function public.set_updated_at();

create trigger articles_set_updated_at
before update on public.articles
for each row execute function public.set_updated_at();

create trigger events_set_updated_at
before update on public.events
for each row execute function public.set_updated_at();

create trigger suggestions_set_updated_at
before update on public.suggestions
for each row execute function public.set_updated_at();
```

## Criação automática do perfil após o cadastro

```sql
create or replace function public.handle_new_user()
returns trigger
language plpgsql
security definer set search_path = public
as $$
begin
  insert into public.users (id, nome, email)
  values (
    new.id,
    coalesce(new.raw_user_meta_data ->> 'nome', split_part(new.email, '@', 1)),
    new.email
  );
  return new;
end;
$$;

create trigger on_auth_user_created
after insert on auth.users
for each row execute function public.handle_new_user();
```

## Row Level Security (RLS)

Ative o RLS antes de conectar o front-end ao Supabase. Estas políticas são uma base inicial e devem ser revisadas antes de produção.

```sql
alter table public.users enable row level security;
alter table public.articles enable row level security;
alter table public.events enable row level security;
alter table public.suggestions enable row level security;
alter table public.media_files enable row level security;
alter table public.audit_logs enable row level security;

create or replace function public.is_admin()
returns boolean
language sql
stable
security definer set search_path = public
as $$
  select exists (
    select 1 from public.users
    where id = auth.uid() and perfil = 'admin' and ativo = true and deleted_at is null
  );
$$;

create policy "public reads published articles"
on public.articles for select
using (status in ('aprovado', 'publicado') and deleted_at is null);

create policy "authenticated authors create articles"
on public.articles for insert to authenticated
with check (author_id = auth.uid());

create policy "authors read own articles"
on public.articles for select to authenticated
using (author_id = auth.uid() or public.is_admin());

create policy "admins manage articles"
on public.articles for all to authenticated
using (public.is_admin())
with check (public.is_admin());

create policy "public reads active events"
on public.events for select
using (status = 'publicado' and deleted_at is null);

create policy "admins manage events"
on public.events for all to authenticated
using (public.is_admin())
with check (public.is_admin());

create policy "anyone sends suggestions"
on public.suggestions for insert
with check (true);

create policy "admins manage suggestions"
on public.suggestions for all to authenticated
using (public.is_admin())
with check (public.is_admin());

create policy "users read own profile"
on public.users for select to authenticated
using (id = auth.uid() or public.is_admin());

create policy "admins manage users"
on public.users for all to authenticated
using (public.is_admin())
with check (public.is_admin());
```

```js
const backup = Object.fromEntries([
  'elza_users',
  'elza_suggestions',
  'elza_articles',
  'elza_events'
```

## Observações para a conexão

1. O front-end usa `SupabaseDatabase` em [WebSite/src/data/database.js](/home/drezin/Documents/Elza/WebSite/src/data/database.js) quando as variáveis `VITE_SUPABASE_URL` e `VITE_SUPABASE_ANON_KEY` existem; sem elas, mantém o fallback local.
2. O cliente está em [supabaseClient.js](/home/drezin/Documents/Elza/WebSite/src/config/supabaseClient.js) e o pacote está declarado em [package.json](/home/drezin/Documents/Elza/WebSite/package.json).
3. O token deve vir de `supabase.auth.getSession()`, não de um token criado manualmente.
4. Nunca coloque a `service_role key` no HTML ou em JavaScript público. No navegador use apenas a chave anon/public.
5. Crie no Supabase Storage o bucket `article-media` para as imagens. O código registra o arquivo em `media_files` e grava a URL em `articles.imagem_url`.
6. A criação de usuários administrativos exige uma Edge Function protegida. O navegador não pode usar a `service_role key`; por isso a tela de administração retorna uma mensagem orientando essa configuração.
7. Para instalar e executar localmente, dentro de [WebSite/](/home/drezin/Documents/Elza/WebSite) use `npm install`, copie `.env.example` para `.env`, preencha as duas variáveis e execute `npm run dev`.

O arquivo `.env` presente atualmente ainda contém valores de exemplo (`seu-projeto` e `sua-chave-anon-public`), portanto o sistema continua corretamente no modo local até você substituí-los pelas credenciais reais.
